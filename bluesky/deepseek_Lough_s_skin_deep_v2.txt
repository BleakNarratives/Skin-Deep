async function getGeminiAnalysis(handData, skinData) {
    // Using AI Studio's Gemini API (free tier)
    const API_KEY = 'YOUR_AI_STUDIO_API_KEY'; // Get free key from https://makersuite.google.com/

        Analyze this tattoo artist data:
        Hand landmarks: ${JSON.stringify(handData)}
        Skin response: ${JSON.stringify(skinData)}
        
        Provide:
        1. Technique analysis (line consistency, pressure variation)
        2. Trauma prediction (overworking risk)
        3. Lough-style similarity score (1-100)
        4. One tip for improvement
        
        Return as JSON.
    `;
    
    try {
        const response = await fetch(`${API_URL}?key=${API_KEY}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                contents: [{
                    parts: [{ text: prompt }]
                }]
            })
        });
        
        const data = await response.json();
        return JSON.parse(data.candidates[0].content.parts[0].text);
    } catch (error) {
        console.error('Gemini API error:', error);
        return getFallbackAnalysis(); // Local fallback
    }
}

const server = http.createServer(app);
const wss = new WebSocket.Server({ server });

// Data structures
const sessions = new Map();
const clients = new Map();

// WebSocket connection for real-time data
wss.on('connection', (ws, req) => {
    const sessionId = req.url.split('/').pop() || 'default';
    const clientId = Date.now() + Math.random().toString(36).substr(2, 9);
    
    clients.set(clientId, ws);
    if (!sessions.has(sessionId)) {
        sessions.set(sessionId, new Set());
    }
    sessions.get(sessionId).add(clientId);
    
    console.log(`Client ${clientId} connected to session ${sessionId}`);
    
    ws.on('message', (message) => {
        try {
            const data = JSON.parse(message);
            handleWebSocketData(clientId, sessionId, data);
            
            // Broadcast to other clients in same session
            sessions.get(sessionId).forEach(id => {
                if (id !== clientId && clients.has(id)) {
                    clients.get(id).send(JSON.stringify({
                        type: 'telemetry',
                        data: data,
                        timestamp: Date.now(),
                        clientId: clientId
                    }));
                }
            });
        } catch (error) {
            console.error('WebSocket message error:', error);
        }
    });
    
    ws.on('close', () => {
        clients.delete(clientId);
        if (sessions.has(sessionId)) {
            sessions.get(sessionId).delete(clientId);
        }
        console.log(`Client ${clientId} disconnected`);
    });
});

        processHandData(data);
    } else if (data.type === 'emg') {
        processEMGData(data);
    } else if (data.type === 'machine_telemetry') {
        processMachineData(data);
    }
}

function processHandData(data) {
    // Calculate hand metrics
    const landmarks = data.landmarks;
    if (landmarks && landmarks.length >= 21) {
        const wrist = landmarks[0];
        const indexTip = landmarks[8];
        
        // Calculate hand stability
        const stability = calculateHandStability(landmarks);
        
        // Calculate trajectory
        const trajectory = calculateTrajectory(landmarks);
        
        // Store in memory cache
        const sessionData = getSessionData(data.sessionId);
        sessionData.handMetrics = {
            stability,
            trajectory,
            wristPosition: wrist,
            indexTipPosition: indexTip,
            timestamp: Date.now()
        };
    }
}

function calculateHandStability(landmarks) {
    // Simple stability calculation based on variance
    const positions = landmarks.map(l => [l.x, l.y, l.z]);
    const mean = positions.reduce((acc, pos) => {
        acc[0] += pos[0];
        acc[1] += pos[1];
        acc[2] += pos[2];
        return acc;
    }, [0, 0, 0]).map(val => val / positions.length);
    
    const variance = positions.reduce((acc, pos) => {
        acc += Math.pow(pos[0] - mean[0], 2) +
               Math.pow(pos[1] - mean[1], 2) +
               Math.pow(pos[2] - mean[2], 2);
        return acc;
    }, 0) / positions.length;
    
    return Math.max(0, 100 - variance * 1000);
}

function calculateTrajectory(landmarks) {
    // Calculate movement direction and speed
    const indexTip = landmarks[8];
    const wrist = landmarks[0];
    
    const dx = indexTip.x - wrist.x;
    const dy = indexTip.y - wrist.y;
    const dz = indexTip.z - wrist.z;
    
    return {
        direction: [dx, dy, dz],
        magnitude: Math.sqrt(dx*dx + dy*dy + dz*dz),
        angle: Math.atan2(dy, dx) * (180 / Math.PI)
    };
}

function calculateOverallStability(data) {
    // Calculate based on hand position variance
    if (!data.telemetry || data.telemetry.length === 0) return 0;
    
    const positions = data.telemetry
        .filter(t => t.type === 'hand_tracking')
        .map(t => t.data.wristPosition);
    
    if (positions.length < 2) return 0;
    
    let totalVariance = 0;
    for (let i = 1; i < positions.length; i++) {
        const dx = positions[i].x - positions[i-1].x;
        const dy = positions[i].y - positions[i-1].y;
        const dz = positions[i].z - positions[i-1].z;
        totalVariance += Math.sqrt(dx*dx + dy*dy + dz*dz);
    }
    
    const avgVariance = totalVariance / (positions.length - 1);
    return Math.max(0, 100 - avgVariance * 100);
}

function calculateLineConsistency(data) {
    // Analyze line consistency from hand paths
    const lines = extractLinesFromTelemetry(data);
    
    if (lines.length === 0) return 0;
    
    let totalConsistency = 0;
    lines.forEach(line => {
        const consistency = analyzeLineConsistency(line);
        totalConsistency += consistency;
    });
    
    return totalConsistency / lines.length;
}

function extractLinesFromTelemetry(data) {
    // Simple line extraction algorithm
    const telemetry = data.telemetry || [];
    const handData = telemetry.filter(t => t.type === 'hand_tracking');
    
    const lines = [];
    let currentLine = [];
    
    for (let i = 0; i < handData.length; i++) {
        const point = handData[i].data.indexTipPosition;
        
        if (currentLine.length === 0 || 
            distance(currentLine[currentLine.length-1], point) < 0.1) {
            currentLine.push(point);
        } else {
            if (currentLine.length > 10) {
                lines.push([...currentLine]);
            }
            currentLine = [point];
        }
    }
    
    if (currentLine.length > 10) {
        lines.push(currentLine);
    }
    
    return lines;
}

function distance(p1, p2) {
    const dx = p1.x - p2.x;
    const dy = p1.y - p2.y;
    const dz = p1.z - p2.z;
    return Math.sqrt(dx*dx + dy*dy + dz*dz);
}

function analyzeLineConsistency(line) {
    // Calculate how straight the line is
    if (line.length < 2) return 0;
    
    // Fit line using linear regression
    const points = line.map(p => ({x: p.x, y: p.y}));
    const n = points.length;
    
    let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
    points.forEach(p => {
        sumX += p.x;
        sumY += p.y;
        sumXY += p.x * p.y;
        sumX2 += p.x * p.x;
    });
    
    const slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
    const intercept = (sumY - slope * sumX) / n;
    
    // Calculate R-squared
    let ssTotal = 0, ssResidual = 0;
    const meanY = sumY / n;
    
    points.forEach(p => {
        const predicted = slope * p.x + intercept;
        ssTotal += Math.pow(p.y - meanY, 2);
        ssResidual += Math.pow(p.y - predicted, 2);
    });
    
    const rSquared = 1 - (ssResidual / ssTotal);
    return Math.max(0, rSquared * 100);
}

function identifyTechniquePatterns(data) {
    const patterns = [];
    const telemetry = data.telemetry || [];
    
    // Identify wrist rotation patterns
    const rotations = detectWristRotations(telemetry);

        patterns.push({

            pattern: analyzePressurePattern(pressureChanges)
        });
    }
    
    return patterns;
}

// ml-processor.js - Runs in browser with TensorFlow.js
class LoughMLProcessor {
    constructor() {
        this.model = null;
        this.isLoaded = false;
        this.bufferSize = 100;
        this.dataBuffer = [];
        this.predictionInterval = null;
    }
    
    async loadModels() {
        console.log('Loading ML models...');
        
        try {
            // Load hand gesture model
            const gestureModel = await this.loadGestureModel();
            
            // Load EMG prediction model
            const emgModel = await this.loadEMGModel();
            
            // Load technique classification model
            const techniqueModel = await this.loadTechniqueModel();
            
            this.models = {
                gesture: gestureModel,
                emg: emgModel,
                technique: techniqueModel
            };
            
            this.isLoaded = true;
            console.log('ML models loaded successfully');
            
        } catch (error) {
            console.error('Failed to load models:', error);
            // Fallback to heuristic methods
            this.useFallback = true;
        }
    }
    
    async loadGestureModel() {
        // Try to load pre-trained model, fallback to custom model
        const modelUrl = 'https://tfhub.dev/mediapipe/handpose/1';
        
        try {
            const model = await tf.loadGraphModel(modelUrl);
            return model;
        } catch (error) {
            console.log('Using local gesture model');
            return await this.createGestureModel();
        }
    }
    
    async createGestureModel() {
        // Create a simple CNN for hand gesture recognition
        const model = tf.sequential();
        
        model.add(tf.layers.conv2d({
            inputShape: [224, 224, 3],
            filters: 32,
            kernelSize: 3,
            activation: 'relu'
        }));
        
        model.add(tf.layers.maxPooling2d({ poolSize: 2 }));
        
        model.add(tf.layers.conv2d({
            filters: 64,
            kernelSize: 3,
            activation: 'relu'
        }));
        
        model.add(tf.layers.maxPooling2d({ poolSize: 2 }));
        
        model.add(tf.layers.flatten());
        model.add(tf.layers.dense({ units: 128, activation: 'relu' }));
        model.add(tf.layers.dense({ units: 10, activation: 'softmax' }));
        
        model.compile({
            optimizer: 'adam',
            loss: 'categoricalCrossentropy',
            metrics: ['accuracy']
        });
        
        return model;
    }
    
    async processHandFrame(handLandmarks, videoFrame) {
        if (!this.isLoaded && !this.useFallback) {
            return this.fallbackProcess(handLandmarks);
        }
        
        try {
            // Process with ML model
            const inputTensor = this.prepareInput(handLandmarks, videoFrame);
            
            let predictions;
            if (this.models && this.models.gesture) {
                predictions = await this.models.gesture.predict(inputTensor);
                inputTensor.dispose();
            } else {
                predictions = this.heuristicProcess(handLandmarks);
            }
            
            return {
                gesture: this.decodeGesture(predictions),
                confidence: predictions.max().dataSync()[0],
                landmarks: handLandmarks,
                timestamp: Date.now()
            };
            
        } catch (error) {
            console.error('ML processing error:', error);
            return this.fallbackProcess(handLandmarks);
        }
    }
    
    prepareInput(landmarks, videoFrame) {
        // Convert landmarks to tensor
        const landmarkArray = [];
        landmarks.forEach(landmark => {
            landmarkArray.push(landmark.x, landmark.y, landmark.z);
        });
        
        return tf.tensor3d([landmarkArray], [1, landmarks.length, 3]);
    }
    
    decodeGesture(predictions) {
        const gestures = [
            'resting',
            'grip_light',
            'grip_medium',
            'grip_heavy',
            'line_pull',
            'shade_circular',
            'shade_whip',
            'wash',
            'stipple',
            'dotwork'
        ];
        
        const predArray = predictions.dataSync();
        const maxIndex = predArray.indexOf(Math.max(...predArray));
        
        return {
            type: gestures[maxIndex],
            index: maxIndex,
            probabilities: predArray
        };
    }
    
    heuristicProcess(landmarks) {
        // Fallback heuristic processing
        const wrist = landmarks[0];
        const indexTip = landmarks[8];
        const thumbTip = landmarks[4];
        
        // Calculate grip strength based on finger proximity
        const gripDistance = this.calculateDistance(indexTip, thumbTip);
        let gripType = 'resting';
        
        if (gripDistance < 0.05) gripType = 'grip_heavy';
        else if (gripDistance < 0.1) gripType = 'grip_medium';
        else if (gripDistance < 0.2) gripType = 'grip_light';
        
        // Detect line pulling motion
        const handDirection = this.calculateHandDirection(landmarks);
        const isPulling = handDirection.y > 0.7 && gripType !== 'resting';
        
        return {
            gesture: isPulling ? 'line_pull' : gripType,
            confidence: 0.85,
            isHeuristic: true
        };
    }
    
    calculateDistance(p1, p2) {
        const dx = p1.x - p2.x;
        const dy = p1.y - p2.y;
        const dz = p1.z - p2.z;
        return Math.sqrt(dx*dx + dy*dy + dz*dz);
    }
    
    calculateHandDirection(landmarks) {
        const wrist = landmarks[0];
        const middleBase = landmarks[9];
        
        return {
            x: middleBase.x - wrist.x,
            y: middleBase.y - wrist.y,
            z: middleBase.z - wrist.z
        };
    }
    
    async predictEMGFromIMU(imuData) {
        // Predict muscle activation from IMU data
        if (!this.models || !this.models.emg) {
            return this.simulateEMG(imuData);
        }
        
        try {
            const input = this.prepareIMUInput(imuData);
            const prediction = await this.models.emg.predict(input);
            input.dispose();
            
            return {
                emg: prediction.dataSync(),
                timestamp: Date.now(),
                confidence: 0.9
            };
        } catch (error) {
            return this.simulateEMG(imuData);
        }
    }
    
    prepareIMUInput(imuData) {
        // Prepare IMU data for neural network
        const features = [
            imuData.acceleration.x, imuData.acceleration.y, imuData.acceleration.z,
            imuData.gyroscope.x, imuData.gyroscope.y, imuData.gyroscope.z,
            imuData.orientation.alpha || 0, imuData.orientation.beta || 0, imuData.orientation.gamma || 0
        ];
        
        return tf.tensor2d([features], [1, features.length]);
    }
    
    simulateEMG(imuData) {
        // Simple EMG simulation based on acceleration
        const accelMagnitude = Math.sqrt(
            imuData.acceleration.x ** 2 +
            imuData.acceleration.y ** 2 +
            imuData.acceleration.z ** 2
        );
        
        const baseEMG = Math.min(1, accelMagnitude / 20);
        const noise = (Math.random() - 0.5) * 0.1;
        
        // Simulate different muscle groups
        return {
            flexorCarpiRadialis: baseEMG * 0.8 + noise,
            extensorDigitorum: baseEMG * 0.6 + noise * 0.8,
            flexorDigitorum: baseEMG * 0.7 + noise * 0.7,
            pronatorTeres: baseEMG * 0.4 + noise * 0.5,
            timestamp: Date.now(),
            isSimulated: true
        };
    }
    
    async classifyTechnique(sequenceData) {
        // Classify tattooing technique from sequence
        if (!this.models || !this.models.technique) {
            return this.heuristicTechniqueClassification(sequenceData);
        }
        
        try {
            const sequenceTensor = this.prepareSequenceInput(sequenceData);
            const classification = await this.models.technique.predict(sequenceTensor);
            sequenceTensor.dispose();
            
            return {
                technique: this.decodeTechnique(classification),
                confidence: classification.max().dataSync()[0],
                features: this.extractTechniqueFeatures(sequenceData)
            };
        } catch (error) {
            return this.heuristicTechniqueClassification(sequenceData);
        }
    }
    
    prepareSequenceInput(sequenceData) {
        // Convert time series data to tensor
        const sequenceLength = Math.min(sequenceData.length, 100);
        const features = 21 * 3; // 21 landmarks * 3 coordinates
        
        const tensorData = new Array(sequenceLength * features).fill(0);
        
        for (let i = 0; i < sequenceLength; i++) {
            const frame = sequenceData[i];
            if (frame && frame.landmarks) {
                for (let j = 0; j < Math.min(21, frame.landmarks.length); j++) {
                    const idx = (i * features) + (j * 3);
                    tensorData[idx] = frame.landmarks[j].x || 0;
                    tensorData[idx + 1] = frame.landmarks[j].y || 0;
                    tensorData[idx + 2] = frame.landmarks[j].z || 0;
                }
            }
        }
        
        return tf.tensor3d([tensorData], [1, sequenceLength, features]);
    }
    
    decodeTechnique(prediction) {
        const techniques = [
            'lining',
            'shading',
            'color_packing',
            'stipple',
            'dotwork',
            'whip_shade',
            'smooth_shade',
            'blackwork',
            'fine_line',
            'bold_line'
        ];
        
        const predArray = prediction.dataSync();
        const maxIndex = predArray.indexOf(Math.max(...predArray));
        
        return {
            type: techniques[maxIndex],
            index: maxIndex,
            probabilities: predArray
        };
    }
    
    heuristicTechniqueClassification(sequenceData) {
        // Simple heuristic technique classification
        if (sequenceData.length < 10) {
            return { technique: 'unknown', confidence: 0 };
        }
        
        // Calculate movement patterns
        const velocities = [];
        for (let i = 1; i < sequenceData.length; i++) {
            if (sequenceData[i].landmarks && sequenceData[i-1].landmarks) {
                const dist = this.calculateDistance(
                    sequenceData[i].landmarks[8],
                    sequenceData[i-1].landmarks[8]
                );
                velocities.push(dist * 100); // Scale up
            }
        }
        
        const avgVelocity = velocities.reduce((a, b) => a + b, 0) / velocities.length;
        const velocityVariance = this.calculateVariance(velocities);
        
        let technique = 'unknown';
        let confidence = 0.5;
        
        if (avgVelocity > 5 && velocityVariance < 2) {
            technique = 'lining';
            confidence = 0.7;
        } else if (avgVelocity < 3 && velocityVariance > 3) {
            technique = 'shading';
            confidence = 0.6;
        } else if (avgVelocity < 2) {
            technique = 'color_packing';
            confidence = 0.65;
        }
        
        return {
            technique: {
                type: technique,
                index: this.getTechniqueIndex(technique),
                probabilities: this.generateProbabilityDistribution(technique)
            },
            confidence,
            features: {
                avgVelocity,
                velocityVariance,
                movementComplexity: this.calculateMovementComplexity(sequenceData)
            }
        };
    }
    
    calculateVariance(array) {
        const mean = array.reduce((a, b) => a + b, 0) / array.length;
        return array.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / array.length;
    }
    
    getTechniqueIndex(technique) {
        const indexMap = {
            'lining': 0,
            'shading': 1,
            'color_packing': 2,
            'stipple': 3,
            'dotwork': 4,
            'whip_shade': 5,
            'smooth_shade': 6,
            'blackwork': 7,
            'fine_line': 8,
            'bold_line': 9
        };
        return indexMap[technique] || 0;
    }
    
    generateProbabilityDistribution(primaryTechnique) {
        const probs = new Array(10).fill(0.01);
        const primaryIndex = this.getTechniqueIndex(primaryTechnique);
        probs[primaryIndex] = 0.8;
        
        // Add some probability to related techniques
        const related = {
            'lining': [8, 9], // fine_line, bold_line
            'shading': [5, 6], // whip_shade, smooth_shade
            'color_packing': [7], // blackwork
            'stipple': [4], // dotwork
            'dotwork': [3] // stipple
        };
        
        if (related[primaryTechnique]) {
            related[primaryTechnique].forEach(idx => {
                probs[idx] = 0.1;
            });
        }
        
        // Normalize
        const sum = probs.reduce((a, b) => a + b, 0);
        return probs.map(p => p / sum);
    }
    
    calculateMovementComplexity(sequenceData) {
        // Calculate fractal dimension of movement
        if (sequenceData.length < 20) return 0;
        
        const positions = sequenceData
            .filter(d => d.landmarks)
            .map(d => ({x: d.landmarks[8].x, y: d.landmarks[8].y}));
        
        if (positions.length < 20) return 0;
        
        // Simple complexity measure based on path length vs straight line distance
        let pathLength = 0;
        for (let i = 1; i < positions.length; i++) {
            pathLength += Math.sqrt(
                Math.pow(positions[i].x - positions[i-1].x, 2) +
                Math.pow(positions[i].y - positions[i-1].y, 2)
            );
        }
        
        const straightDistance = Math.sqrt(
            Math.pow(positions[positions.length-1].x - positions[0].x, 2) +
            Math.pow(positions[positions.length-1].y - positions[0].y, 2)
        );
        
        return straightDistance > 0 ? pathLength / straightDistance : 1;
    }
    
    startRealTimePrediction(callback, interval = 100) {
        if (this.predictionInterval) {
            clearInterval(this.predictionInterval);
        }
        
        this.predictionInterval = setInterval(async () => {
            if (this.dataBuffer.length > 0) {
                const latestData = this.dataBuffer[this.dataBuffer.length - 1];
                const prediction = await this.processHandFrame(
                    latestData.landmarks,
                    latestData.videoFrame
                );
                
                if (callback) {
                    callback(prediction);
                }
            }
        }, interval);
    }
    
    stopRealTimePrediction() {
        if (this.predictionInterval) {
            clearInterval(this.predictionInterval);
            this.predictionInterval = null;
        }
    }
    
    addToBuffer(data) {
        this.dataBuffer.push(data);
        if (this.dataBuffer.length > this.bufferSize) {
            this.dataBuffer.shift();
        }
    }
    
    clearBuffer() {
        this.dataBuffer = [];
    }
    
    getBufferStats() {
        return {
            size: this.dataBuffer.length,
            duration: this.dataBuffer.length * (1000 / 30), // Assuming 30fps
            memoryUsage: this.estimateMemoryUsage()
        };
    }
    
    estimateMemoryUsage() {
        const avgSize = 1024; // Rough average size per entry in bytes
        return (this.dataBuffer.length * avgSize) / 1024; // KB
    }
}

        if (depth > 0.1 && Math.random() < 0.3) {

    createInkParticle() {

    updateParticles() {
        // Update ink particles
        for (let i = this.inkParticles.length - 1; i >= 0; i--) {
            const particle = this.inkParticles[i];
            particle.life -= 1;
            
            if (particle.life <= 0) {
                this.scene.remove(particle.mesh);
                this.inkParticles.splice(i, 1);
                continue;
            }
            
            // Update position
            particle.mesh.position.add(particle.velocity);
            
            // Update opacity
            const opacity = particle.life / 100;
            particle.mesh.material.opacity = opacity * 0.8;
            
            // Sink into skin
            particle.mesh.position.y -= 0.001;
        }
        
        // Update trauma zones
        for (let i = this.traumaZones.length - 1; i >= 0; i--) {
            const zone = this.traumaZones[i];
            zone.life -= 1;
            
            if (zone.life <= 0) {
                this.scene.remove(zone.mesh);
                this.traumaZones.splice(i, 1);
                continue;
            }
            
            // Update opacity and scale
            const lifeRatio = zone.life / zone.maxLife;
            zone.mesh.material.opacity = lifeRatio * 0.3;
            zone.mesh.scale.setScalar(1 + (1 - lifeRatio) * 0.5);
        }
    }

        this.animationFrame = requestAnimationFrame(() => this.animate());
        
        // Update particles
        this.updateParticles();

class LoughDataSync {
    constructor(options = {}) {
        this.options = {
            serverUrl: 'ws://localhost:3000',
            sessionId: 'default',
            reconnectInterval: 5000,
            maxRetries: 10,
            compression: true,
            batchSize: 50,
            ...options
        };
        
        this.ws = null;
        this.isConnected = false;
        this.reconnectAttempts = 0;
        this.dataQueue = [];
        this.batchInterval = null;
        this.eventListeners = new Map();
        
        this.metrics = {
            bytesSent: 0,
            bytesReceived: 0,
            messagesSent: 0,
            messagesReceived: 0,
            latency: 0
        };
        
        this.init();
    }
    
    init() {
        this.connect();
        this.startBatchProcessor();
    }
    
    connect() {
        try {
            this.ws = new WebSocket(`${this.options.serverUrl}/ws/${this.options.sessionId}`);
            
            this.ws.onopen = () => {
                console.log('WebSocket connected');
                this.isConnected = true;
                this.reconnectAttempts = 0;
                this.emit('connected');
                
                // Send queued data
                this.processQueue();
            };
            
            this.ws.onmessage = (event) => {
                this.handleMessage(event.data);
            };
            
            this.ws.onclose = () => {
                console.log('WebSocket disconnected');
                this.isConnected = false;
                this.emit('disconnected');
                this.attemptReconnect();
            };
            
            this.ws.onerror = (error) => {
                console.error('WebSocket error:', error);
                this.emit('error', error);
            };
            
        } catch (error) {
            console.error('Failed to connect:', error);
            this.attemptReconnect();
        }
    }
    
    attemptReconnect() {
        if (this.reconnectAttempts < this.options.maxRetries) {
            this.reconnectAttempts++;
            console.log(`Reconnecting in ${this.options.reconnectInterval}ms (attempt ${this.reconnectAttempts})`);
            
            setTimeout(() => {
                this.connect();
            }, this.options.reconnectInterval);
        } else {
            console.error('Max reconnection attempts reached');
            this.emit('connection_failed');
        }
    }
    
    send(data, type = 'telemetry') {
        const message = {
            type,
            data,
            sessionId: this.options.sessionId,
            timestamp: Date.now(),
            clientId: this.getClientId()
        };
        
        if (this.options.compression) {
            this.compressMessage(message).then(compressed => {
                this.queueMessage(compressed);
            });
        } else {
            this.queueMessage(message);
        }
    }
    
    async compressMessage(message) {
        // Simple compression for large data
        if (typeof message === 'object') {
            const jsonString = JSON.stringify(message);
            
            // Use TextEncoder for compression
            const encoder = new TextEncoder();
            const data = encoder.encode(jsonString);
            
            // Simple run-length encoding for repeated values
            const compressed = this.runLengthEncode(data);
            
            return {
                compressed: true,
                data: Array.from(compressed),
                originalSize: jsonString.length,
                compressedSize: compressed.length
            };
        }
        
        return message;
    }
    
    runLengthEncode(data) {
        const encoded = [];
        let count = 1;
        
        for (let i = 1; i <= data.length; i++) {
            if (i < data.length && data[i] === data[i-1]) {
                count++;
            } else {
                encoded.push(count, data[i-1]);
                count = 1;
            }
        }
        
        return new Uint8Array(encoded);
    }
    
    queueMessage(message) {
        this.dataQueue.push(message);
        this.metrics.bytesSent += JSON.stringify(message).length;
        
        // If queue is getting large, process immediately
        if (this.dataQueue.length >= this.options.batchSize * 2) {
            this.processQueue();
        }
    }
    
    startBatchProcessor() {
        this.batchInterval = setInterval(() => {
            if (this.dataQueue.length > 0) {
                this.processQueue();
            }
        }, 1000); // Process every second
    }
    
    processQueue() {
        if (!this.isConnected || this.dataQueue.length === 0) {
            return;
        }
        
        // Take batch from queue
        const batch = this.dataQueue.splice(0, this.options.batchSize);
        
        if (batch.length > 0) {
            const batchMessage = {
                type: 'batch',
                data: batch,
                count: batch.length,
                timestamp: Date.now()
            };
            
            this.sendRaw(batchMessage);
            this.metrics.messagesSent += batch.length;
        }
    }
    
    sendRaw(message) {
        if (this.ws && this.isConnected) {
            try {
                this.ws.send(JSON.stringify(message));
            } catch (error) {
                console.error('Failed to send message:', error);
                // Requeue failed messages
                if (message.type === 'batch') {
                    this.dataQueue.unshift(...message.data);
                } else {
                    this.dataQueue.unshift(message);
                }
            }
        }
    }
    
    handleMessage(rawData) {
        const startTime = Date.now();
        
        try {
            let data;
            
            if (typeof rawData === 'string') {
                data = JSON.parse(rawData);
            } else if (rawData instanceof ArrayBuffer) {
                data = this.decompressBinary(rawData);
            } else {
                console.error('Unknown message format:', rawData);
                return;
            }
            
            this.metrics.bytesReceived += rawData.length;
            this.metrics.messagesReceived++;
            
            // Calculate latency
            if (data.timestamp) {
                this.metrics.latency = Date.now() - data.timestamp;
            }
            
            // Handle different message types
            switch(data.type) {
                case 'telemetry':
                    this.emit('telemetry', data.data);
                    break;
                    
                case 'batch':
                    data.data.forEach(item => {
                        this.emit(item.type || 'data', item.data);
                    });
                    break;
                    
                case 'command':
                    this.handleCommand(data.command, data.payload);
                    break;
                    
                case 'analysis':
                    this.emit('analysis', data.data);
                    break;
                    
                case 'error':
                    console.error('Server error:', data.message);
                    this.emit('error', data);
                    break;
                    
                default:
                    this.emit('message', data);
            }
            
        } catch (error) {
            console.error('Failed to process message:', error, rawData);
            this.emit('parse_error', { error, rawData });
        }
    }
    
    decompressBinary(binaryData) {
        // Decompress run-length encoded data
        const data = new Uint8Array(binaryData);
        const decoded = [];
        
        for (let i = 0; i < data.length; i += 2) {
            const count = data[i];
            const value = data[i + 1];
            
            for (let j = 0; j < count; j++) {
                decoded.push(value);
            }
        }
        
        const decoder = new TextDecoder();
        const jsonString = decoder.decode(new Uint8Array(decoded));
        return JSON.parse(jsonString);
    }
    
    handleCommand(command, payload) {
        switch(command) {
            case 'start_recording':
                this.emit('start_recording', payload);
                break;
                
            case 'stop_recording':
                this.emit('stop_recording', payload);
                break;
                
            case 'clear_data':
                this.dataQueue = [];
                this.emit('data_cleared');
                break;
                
            case 'get_metrics':
                this.sendMetrics();
                break;
                
            case 'change_session':
                this.options.sessionId = payload.sessionId;
                this.reconnect();
                break;
        }
    }
    
    sendMetrics() {
        this.send({
            type: 'metrics',
            data: this.metrics
        }, 'command');
    }
    
    on(event, callback) {
        if (!this.eventListeners.has(event)) {
            this.eventListeners.set(event, []);
        }
        this.eventListeners.get(event).push(callback);
    }
    
    off(event, callback) {
        if (this.eventListeners.has(event)) {
            const listeners = this.eventListeners.get(event);
            const index = listeners.indexOf(callback);
            if (index > -1) {
                listeners.splice(index, 1);
            }
        }
    }
    
    emit(event, data) {
        if (this.eventListeners.has(event)) {
            this.eventListeners.get(event).forEach(callback => {
                try {
                    callback(data);
                } catch (error) {
                    console.error(`Error in event listener for ${event}:`, error);
                }
            });
        }
    }
    
    getClientId() {
        if (!this.clientId) {
            this.clientId = 'client_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
        }
        return this.clientId;
    }
    
    reconnect() {
        if (this.ws) {
            this.ws.close();
        }
        this.connect();
    }
    
    disconnect() {
        if (this.batchInterval) {
            clearInterval(this.batchInterval);
        }
        
        if (this.ws) {
            this.ws.close();
        }
        
        this.isConnected = false;
        this.emit('disconnected');
    }
    
    getQueueSize() {
        return this.dataQueue.length;
    }
    
    clearQueue() {
        this.dataQueue = [];
    }
    
    getMetrics() {
        return {
            ...this.metrics,
            queueSize: this.dataQueue.length,
            connected: this.isConnected,
            reconnectAttempts: this.reconnectAttempts
        };
    }
}

const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');
const archiver = require('archiver');
const fs = require('fs-extra');
const path = require('path');
const { format } = require('date-fns');

console.log('💾 LOUGH Backup Script');
console.log('=====================');

class LoughBackup {
    constructor() {
        this.backupDir = './backups';
        this.dataDir = './data';
        this.maxBackups = 10;
        
        // Ensure backup directory exists
        fs.ensureDirSync(this.backupDir);
    }
    
    async createBackup() {
        const timestamp = format(new Date(), 'yyyy-MM-dd_HH-mm-ss');
        const backupName = `lough_backup_${timestamp}`;
        const backupPath = path.join(this.backupDir, backupName);
        
        console.log(`📦 Creating backup: ${backupName}`);
        
        try {
            // Create backup directory
            await fs.ensureDir(backupPath);
            
            // Copy data directory
            await this.copyDataDirectory(backupPath);
            
            // Create metadata
            await this.createMetadata(backupPath, backupName);
            
            // Compress backup
            const zipPath = await this.compressBackup(backupPath);
            
            // Clean up temporary directory
            await fs.remove(backupPath);
            
            // Remove old backups if exceeding limit
            await this.cleanupOldBackups();
            
            console.log(`✅ Backup created: ${zipPath}`);
            return zipPath;
            
        } catch (error) {
            console.error('❌ Backup failed:', error);
            throw error;
        }
    }
    
    async copyDataDirectory(backupPath) {
        const dataBackupPath = path.join(backupPath, 'data');
        
        // Copy entire data directory
        await fs.copy(this.dataDir, dataBackupPath);
        
        console.log(`📁 Copied data directory (${await this.getDirectorySize(this.dataDir)} MB)`);
    }
    
    async createMetadata(backupPath, backupName) {
        const metadata = {
            name: backupName,
            timestamp: new Date().toISOString(),
            version: require('../package.json').version,
            dataSize: await this.getDirectorySize(this.dataDir),
            sessionCount: await this.countSessions(),
            recordingCount: await this.countRecordings()
        };
        
        const metadataPath = path.join(backupPath, 'metadata.json');
        await fs.writeJson(metadataPath, metadata, { spaces: 2 });
        
        console.log('📄 Created backup metadata');
    }
    
    async compressBackup(backupPath) {
        const zipPath = `${backupPath}.zip`;
        
        return new Promise((resolve, reject) => {
            const output = fs.createWriteStream(zipPath);
            const archive = archiver('zip', { zlib: { level: 9 } });
            
            output.on('close', () => {
                console.log(`📦 Compressed backup (${(archive.pointer() / 1024 / 1024).toFixed(2)} MB)`);
                resolve(zipPath);
            });
            
            archive.on('error', reject);
            
            archive.pipe(output);
            archive.directory(backupPath, false);
            archive.finalize();
        });
    }
    
    async cleanupOldBackups() {
        const files = await fs.readdir(this.backupDir);
        const backupFiles = files
            .filter(f => f.endsWith('.zip'))
            .map(f => ({
                name: f,
                path: path.join(this.backupDir, f),
                time: fs.statSync(path.join(this.backupDir, f)).mtime.getTime()
            }))
            .sort((a, b) => b.time - a.time);
        
        // Remove backups beyond maxBackups limit
        if (backupFiles.length > this.maxBackups) {
            const toRemove = backupFiles.slice(this.maxBackups);
            
            for (const file of toRemove) {
                await fs.remove(file.path);
                console.log(`🗑️  Removed old backup: ${file.name}`);
            }
        }
    }
    
    async listBackups() {
        const files = await fs.readdir(this.backupDir);
        const backupFiles = files
            .filter(f => f.endsWith('.zip'))
            .map(f => {
                const stat = fs.statSync(path.join(this.backupDir, f));
                return {
                    name: f,
                    size: (stat.size / 1024 / 1024).toFixed(2) + ' MB',
                    modified: stat.mtime
                };
            })
            .sort((a, b) => b.modified - a.modified);
        
        return backupFiles;
    }
    
    async restoreBackup(backupName) {
        const backupPath = path.join(this.backupDir, backupName);
        
        if (!await fs.pathExists(backupPath)) {
            throw new Error(`Backup not found: ${backupName}`);
        }
        
        console.log(`🔧 Restoring backup: ${backupName}`);
        
        try {
            // Extract backup
            const tempDir = path.join(this.backupDir, 'temp_restore');
            await this.extractBackup(backupPath, tempDir);
            
            // Validate backup
            await this.validateBackup(tempDir);
            
            // Stop server if running (you'd need server control logic here)
            console.log('⚠️  Stopping server... (implement server stop)');
            
            // Backup current data

            throw new Error('Invalid backup: missing metadata');
        }
        
        if (!await fs.pathExists(dataPath)) {
            throw new Error('Invalid backup: missing data directory');
        }
        
        console.log('✅ Backup validation passed');
    }

        const sourceData = path.join(tempDir, 'data');
        
        // Remove current data

        // Copy backup data
        await fs.copy(sourceData, this.dataDir);
        
        console.log('📁 Restored data directory');
    }

        if (!await fs.pathExists(dirPath)) return 0;

            if (stats.isDirectory()) {
                const files = await fs.readdir(path);

                    files.map(file => getSize(path.join(path, file)))
